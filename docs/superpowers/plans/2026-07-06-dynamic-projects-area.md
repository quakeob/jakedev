# Dynamic Projects Area Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add build-time git activity data (sparklines, last-active, git-strip), filter chips, and motion to the jakedev.org homepage projects list.

**Architecture:** A dependency-free Node script collects git stats (local repos + GitHub API) into `src/data/activity.json` before every build. `index.astro` joins that data to cards by a new `id` field and renders sparklines/attrs server-side; filtering, relative times, sparkline draw-in, and typing effects are small vanilla-JS additions to the page's existing inline script. Styles follow the repo's conventions (dark top-level, light overrides nested in `:root:not(.dark)`).

**Tech Stack:** Astro 5, vanilla JS, Node ≥18 (global fetch), no new dependencies.

**Spec:** `docs/superpowers/specs/2026-07-06-dynamic-projects-area-design.md`

```js
#!/usr/bin/env node
// Collects git activity for jakedev.org project cards into src/data/activity.json.
// Runs as `prebuild`. Must never fail the build: per-project errors keep the
// previous entry. No dependencies; Node >= 18.
import { execSync } from 'node:child_process';
import { readFileSync, writeFileSync, mkdirSync } from 'node:fs';
import { homedir } from 'node:os';
import { fileURLToPath } from 'node:url';
import path from 'node:path';

const OUT = fileURLToPath(new URL('../src/data/activity.json', import.meta.url));

// slug -> git source. showMessage: false hides the commit-message line on the site.
const SOURCES = {
  'fuel':               { github: 'quakeob/fuel-ios' },
  'vault-keeper':       { github: 'quakeob/ObsidianVaultkeeper' },
  'todolock':           { local: '~/Projects/ScreenLockTodo' },
  'token-tracker':      { local: '~/Projects/ClaudeTokenTracker' },
  'missionary-moments': { local: '~/Code/missionarymoments' },
  'lumice':             { local: '~/Code/lumice' },
  'slumbr':             { local: '~/Code/Slumbr' },
  'uesc-monitor':       { local: '~/Projects/uesc-gg' },
  // slopscore, yerb, marketplace-scraper, neural-networks, time-audit, caverun:
  // no reachable git history — cards fall back to status text.
};

const WEEKS = 12;

function bucketWeekly(epochSeconds) {
  const now = Date.now() / 1000;
  const weekly = new Array(WEEKS).fill(0);
  for (const t of epochSeconds) {
    const weeksAgo = Math.floor((now - t) / (7 * 86400));
    if (weeksAgo >= 0 && weeksAgo < WEEKS) weekly[WEEKS - 1 - weeksAgo]++;
  }
  return weekly;
}

function collectLocal(dir) {
  const cwd = dir.replace(/^~/, homedir());
  const run = (cmd) => execSync(cmd, { cwd, encoding: 'utf8', stdio: ['ignore', 'pipe', 'ignore'] }).trim();
  const lastCommit = run('git log -1 --format=%cI');
  const commitCount = parseInt(run('git rev-list --count HEAD'), 10);
  const lastMessage = run('git log -1 --format=%s');
  const ts = run(`git log --since="${WEEKS * 7} days ago" --format=%ct`)
    .split('\n').filter(Boolean).map(Number);
  return { lastCommit, commitCount, lastMessage, weekly: bucketWeekly(ts) };
}

async function gh(pathname) {
  const res = await fetch(`https://api.github.com${pathname}`, {
    headers: { 'User-Agent': 'jakedev-activity', 'Accept': 'application/vnd.github+json' },
  });
  if (!res.ok) throw new Error(`GET ${pathname} -> ${res.status}`);
  return res;
}

async function collectGithub(repo) {
  const meta = await (await gh(`/repos/${repo}`)).json();
  const commitsRes = await gh(`/repos/${repo}/commits?per_page=1`);
  const link = commitsRes.headers.get('link') || '';
  const m = link.match(/[?&]page=(\d+)>; rel="last"/);
  const commitCount = m ? parseInt(m[1], 10) : 1;
  const [latest] = await commitsRes.json();
  const lastMessage = (latest?.commit?.message || '').split('\n')[0];
  // stats/participation returns 202 while GitHub computes it; treat as no data.
  let weekly = new Array(WEEKS).fill(0);
  try {
    const partRes = await gh(`/repos/${repo}/stats/participation`);
    if (partRes.status === 200) {
      const part = await partRes.json();
      if (Array.isArray(part.all)) weekly = part.all.slice(-WEEKS);
      while (weekly.length < WEEKS) weekly.unshift(0);
    }
  } catch { /* keep zeros */ }
  return { lastCommit: meta.pushed_at, commitCount, lastMessage, weekly };
}

let previous = { projects: {} };
try { previous = JSON.parse(readFileSync(OUT, 'utf8')); } catch { /* first run */ }

const projects = {};
for (const [slug, src] of Object.entries(SOURCES)) {
  try {
    const data = src.local ? collectLocal(src.local) : await collectGithub(src.github);
    projects[slug] = { ...data, showMessage: src.showMessage !== false };
    console.log(`  ok ${slug}: ${data.commitCount} commits, last ${data.lastCommit}`);
  } catch (err) {
    const prev = previous.projects?.[slug];
    if (prev) {
      projects[slug] = prev;
      console.warn(`  warn ${slug}: ${err.message} — kept previous entry`);
    } else {
      console.warn(`  warn ${slug}: ${err.message} — no data`);
    }
  }
}

mkdirSync(path.dirname(OUT), { recursive: true });
writeFileSync(OUT, JSON.stringify({ generated: new Date().toISOString(), projects }, null, 2) + '\n');
console.log(`activity.json written: ${Object.keys(projects).length} projects`);
```

- [ ] **Step 1: Append the projects-dynamics IIFE**

```js
    // ═══════════ PROJECTS: LIVE DATA + FILTERS + MOTION ═══════════
    (function() {
      const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
      const cards = Array.from(document.querySelectorAll('.project-list .project-card'));
      if (!cards.length) return;

      function relTime(iso) {
        const s = (Date.now() - Date.parse(iso)) / 1000;
        if (s < 3600) return Math.max(1, Math.floor(s / 60)) + 'm ago';
        if (s < 86400) return Math.floor(s / 3600) + 'h ago';
        if (s < 604800) return Math.floor(s / 86400) + 'd ago';
        if (s < 86400 * 30) return Math.floor(s / 604800) + 'w ago';
        if (s < 86400 * 365) return Math.floor(s / (86400 * 30)) + 'mo ago';
        return Math.floor(s / (86400 * 365)) + 'y ago';
      }
      const isFresh = (iso) => Date.now() - Date.parse(iso) < 7 * 86400e3;

      // Relative times: card dots + git-strips
      document.querySelectorAll('.activity-last[data-last]').forEach((el) => {
        el.textContent = '● ' + relTime(el.dataset.last);
        if (isFresh(el.dataset.last)) el.classList.add('is-fresh');
      });
      document.querySelectorAll('.git-strip').forEach((strip) => {
        const active = strip.querySelector('.git-strip__active');
        if (active && active.dataset.last) {
          active.textContent = 'active ' + relTime(active.dataset.last);
          if (isFresh(active.dataset.last)) strip.querySelector('.git-strip__dot').classList.add('is-fresh');
        }
      });

      // Heading stat
      const stat = document.getElementById('projects-stat');
      if (stat && stat.dataset.lastPush) {
        stat.textContent = cards.length + ' projects · last push ' + relTime(stat.dataset.lastPush);
      }

      // Sparkline draw-in
      const sparks = document.querySelectorAll('.activity-spark');
      if (reduceMotion) {
        sparks.forEach((s) => s.classList.add('is-drawn'));
      } else {
        const sparkObs = new IntersectionObserver((entries) => {
          entries.forEach((e) => {
            if (e.isIntersecting) { e.target.classList.add('is-drawn'); sparkObs.unobserve(e.target); }
          });
        }, { threshold: 0.4 });
        sparks.forEach((s) => sparkObs.observe(s));
      }

      // Git-strip typing (once per card)
      cards.forEach((card) => {
        const log = card.querySelector('.git-strip__log[data-message]');
        if (!log) return;
        const target = log.querySelector('.git-strip__typed');
        const msg = log.dataset.message;
        if (reduceMotion) { target.textContent = msg; return; }
        let typed = false;
        card.addEventListener('mouseenter', () => {
          if (typed) return;
          typed = true;
          let i = 0;
          const tick = setInterval(() => {
            target.textContent = msg.slice(0, ++i);
            if (i >= msg.length) clearInterval(tick);
          }, 20);
        });
      });

      // Filter chips
      const chips = document.querySelectorAll('#project-filters .project-chip');
      function matches(card, chip) {
        if (chip === 'all') return true;
        if (chip === 'active') {
          const iso = card.dataset.lastCommit;
          return !!iso && Date.now() - Date.parse(iso) < 30 * 86400e3;
        }
        return (card.dataset.chips || '').split(' ').includes(chip);
      }
      // Fill counts
      chips.forEach((chipEl) => {
        const n = cards.filter((c) => matches(c, chipEl.dataset.chip)).length;
        chipEl.querySelector('.project-chip__count').textContent = String(n);
      });
      function applyFilter(chip) {
        let n = 0;
        cards.forEach((card) => {
          const show = matches(card, chip);
          if (show) {
            n++;
            card.querySelector('.project-card__index').textContent = String(n).padStart(2, '0');
            card.style.display = '';
            if (reduceMotion) card.classList.remove('is-filtered-out');
            else requestAnimationFrame(() => requestAnimationFrame(() => card.classList.remove('is-filtered-out')));
          } else {
            card.classList.add('is-filtered-out');
            if (reduceMotion) card.style.display = 'none';
            else setTimeout(() => { if (card.classList.contains('is-filtered-out')) card.style.display = 'none'; }, 320);
          }
        });
      }
      chips.forEach((chipEl) => {
        chipEl.addEventListener('click', () => {
          chips.forEach((c) => c.classList.remove('is-active'));
          chipEl.classList.add('is-active');
          applyFilter(chipEl.dataset.chip);
        });
      });
    })();
```
